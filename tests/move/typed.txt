# Typed at COMMAND.COM's prompt by scripts/move-test when tests/move/mv.bat
# ends, a line at each prompt: commands not run from a batch file, which
# MOVE asks about by default. "== keys KEYS" lines give the answers to the
# questions that follow, and are not typed.

EL

EL

EL

EL

EL

EL
COPY NEWD\X.DAT SUB > NUL

EL

EL
